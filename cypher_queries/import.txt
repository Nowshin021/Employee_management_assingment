CREATE CONSTRAINT IF NOT EXISTS FOR (p:Position) REQUIRE p.positionId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (u:Unit) REQUIRE u.unitCode IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (uo:UnitOffering) REQUIRE uo.unitOfferingId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (ec:EmploymentContract) REQUIRE ec.contractId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (q:Qualification) REQUIRE q.qualId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (l:Leave) REQUIRE l.leaveId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (t:Timesheet) REQUIRE t.timesheetId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (pr:PayRecord) REQUIRE pr.payId IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (rev:PerformanceReview) REQUIRE rev.reviewId IS UNIQUE;

// --- 2.1 Faculty Nodes ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Faculty.csv' AS row
MERGE (f:Faculty {facultyId: toInteger(row.facultyId)})
ON CREATE SET f.name = trim(row.name);

// --- 2.2 Department Nodes ---

// 3. CONNECT RELATIONSHIPS
// ==============================================================================

// --- 3.1 Department belongs to Faculty ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Department.csv' AS row
MATCH (d:Department {deptId: toInteger(row.deptId)})
MATCH (f:Faculty {facultyId: toInteger(row.facultyId)})
MERGE (d)-[:BELONGS_TO]->(f);

// --- 3.2 Employee works in Department and is located at Campus ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Employee.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (d:Department {deptId: toInteger(row.deptId)})
MATCH (c:Campus {campusId: toInteger(row.campusId)})
MERGE (e)-[:WORKS_IN]->(d)
MERGE (e)-[:LOCATED_AT]->(c);

// --- 3.2.1 Employee reports to Manager (self-referencing hierarchy) ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Employee.csv' AS row
WITH row WHERE row.managerId IS NOT NULL
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (m:Employee {staffId: toInteger(row.managerId)})
MERGE (e)-[:REPORTS_TO]->(m);

// --- 3.3 Employee holds Position (Junction table) ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/EmployeePosition.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (p:Position {positionId: toInteger(row.positionId)})
MERGE (e)-[r:HOLDS_POSITION]->(p)
ON CREATE SET r.assignmentId = toInteger(row.assignmentId),
              r.startDate = row.startDate,
              r.endDate = row.endDate,
              r.isPrimary = toInteger(row.isPrimary);

// --- 3.4 Employee has Employment Contract ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/EmploymentContract.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (ec:EmploymentContract {contractId: toInteger(row.contractId)})
MERGE (e)-[:HAS_CONTRACT]->(ec);

// --- 3.5 Employee achieved Qualification ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Qualification.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (q:Qualification {qualId: toInteger(row.qualId)})
MERGE (e)-[:ACHIEVED]->(q);

// --- 3.6 Unit Offering relationships (offers Unit, delivered at Campus) ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/UnitOffering.csv' AS row
MATCH (uo:UnitOffering {unitOfferingId: toInteger(row.unitOfferingId)})
MATCH (u:Unit {unitCode: trim(row.unitCode)})
MATCH (c:Campus {campusId: toInteger(row.campusId)})
MERGE (uo)-[:OFFERS_UNIT]->(u)
MERGE (uo)-[:DELIVERED_AT]->(c);

// --- 3.6.1 Unit owned by Department (School) ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Unit.csv' AS row
MATCH (u:Unit {unitCode: trim(row.unitCode)})
MATCH (d:Department {deptId: toInteger(row.deptId)})
MERGE (u)-[:OWNED_BY]->(d);

// --- 3.6.2 Academic staff coordinates Unit Offering ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/UnitOffering.csv' AS row
MATCH (uo:UnitOffering {unitOfferingId: toInteger(row.unitOfferingId)})
MATCH (coord:AcademicStaff {staffId: toInteger(row.coordinatorId)})
MERGE (coord)-[:COORDINATES]->(uo);

// --- 3.7 Timesheet submitted by Casual Sessional Staff and logged for Unit Offering ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/TimeSheet.csv' AS row
MATCH (e:CasualSessionalStaff {staffId: toInteger(row.staffId)})
MATCH (t:Timesheet {timesheetId: toInteger(row.timesheetID)})
MERGE (e)-[:SUBMITTED]->(t)
WITH row, t
WHERE row.unitOfferingId IS NOT NULL
MATCH (uo:UnitOffering {unitOfferingId: toInteger(row.unitOfferingId)})
MERGE (t)-[:FOR_UNIT_OFFERING]->(uo);

// --- 3.8 Employee requested Leave ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/Leave.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (l:Leave {leaveId: toInteger(row.leaveId)})
MERGE (e)-[:REQUESTED]->(l);

// --- 3.9 Employee issued Pay Record ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/PayRecord.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (pr:PayRecord {payId: toInteger(row.payId)})
MERGE (e)-[:ISSUED_PAY]->(pr);

// --- 3.10 Performance Review (Received review and Reviewer who conducted it) ---
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Nowshin021/Employee_management_assingment/nazia/csv_files/PerformanceReview.csv' AS row
MATCH (e:Employee {staffId: toInteger(row.staffId)})
MATCH (rev:PerformanceReview {reviewId: toInteger(row.reviewId)})
MERGE (e)-[:RECEIVED_REVIEW]->(rev)
WITH row, rev
WHERE row.reviewerId IS NOT NULL
MATCH (reviewer:Employee {staffId: toInteger(row.reviewerId)})
MERGE (reviewer)-[:CONDUCTED_REVIEW]->(rev);
